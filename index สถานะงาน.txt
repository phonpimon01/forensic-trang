<script>
  // 🔗 ใส่ Web App URL ของ Google Apps Script ของคุณเรียบร้อยแล้ว
  const GAS_API_URL = 'https://script.google.com/macros/s/AKfycbxDz5nTAk0R_rKJyiXVFY6rxA2NIeCuF2v5HicpLEyLiQLqzMqzeV114lL7czppJfwyjQ/exec?action=getData';

  let rawData = [];
  let filteredData = [];
  let currentFormattedMessage = "";
  const CACHE_KEY = 'forensic_trang_vercel_v1';

  const Notify = {
    copySuccess: function(msg = 'คัดลอกข้อความสรุปเรียบร้อยแล้ว!') {
      Swal.fire({ icon: 'success', title: msg, toast: true, position: 'top', showConfirmButton: false, timer: 1800, timerProgressBar: true });
    },
    error: function(title, text) { Swal.fire({ icon: 'error', title: title, text: text, confirmButtonText: 'ปิด' }); },
    warning: function(title, text) { Swal.fire({ icon: 'warning', title: title, text: text, confirmButtonText: 'รับทราบ' }); },
    loading: function(title = 'กำลังโหลดข้อมูล...') { Swal.fire({ title: title, allowOutsideClick: false, didOpen: () => { Swal.showLoading(); } }); },
    close: function() { Swal.close(); }
  };

  window.onload = () => {
    Notify.loading('กำลังเชื่อมต่อฐานข้อมูล...');
    const cached = localStorage.getItem(CACHE_KEY);
    if (cached) {
      const p = JSON.parse(cached);
      rawData = p.data;
      setupStations(p.stations);
      updateOverviewStats(rawData);
      Notify.close();
      runSearch();
    }
    updateData();
  };

  function updateData() {
    fetch(GAS_API_URL)
      .then(res => res.json())
      .then(res => {
        localStorage.setItem(CACHE_KEY, JSON.stringify(res));
        rawData = res.data;
        setupStations(res.stations);
        document.getElementById('syncStatus').innerText = 'อัปเดตข้อมูลล่าสุดเรียบร้อย: ' + res.syncTime;
        updateOverviewStats(rawData);
        Notify.close();
        runSearch();
      })
      .catch(err => {
        console.error(err);
        Notify.close();
        Notify.error('ดึงข้อมูลไม่สำเร็จ', 'ไม่สามารถเชื่อมต่อฐานข้อมูล Google Sheets ได้ กรุณาตรวจสอบการอนุญาตสิทธิ์ Web App (Anyone)');
      });
  }

  function updateOverviewStats(data) {
    document.getElementById('statTotalCount').innerHTML = `${data.length} <span class="fs-6 fw-normal text-muted">รายการ</span>`;
    const done = data.filter(d => getStatusInfo(d.st).id === "1" || getStatusInfo(d.st).id === "3").length;
    const pending = data.filter(d => getStatusInfo(d.st).id === "2").length;
    document.getElementById('statDoneCount').innerHTML = `${done} <span class="fs-6 fw-normal text-muted">รายการ</span>`;
    document.getElementById('statPendingCount').innerHTML = `${pending} <span class="fs-6 fw-normal text-muted">รายการ</span>`;
  }

  function setupStations(stations) {
    const select = document.getElementById('stationSelect');
    select.innerHTML = '<option value="">ทุก สภ. ในจังหวัดตรัง</option>';
    stations.forEach(s => {
      let opt = document.createElement('option');
      opt.value = s; opt.text = `สภ.${s}`;
      select.appendChild(opt);
    });
  }

  function getStatusInfo(st) {
    if (st.includes('เสร็จ') && !st.includes('รับผล')) return { id: "1", label: 'เสร็จแล้ว', tag: 'tag-1', icon: 'check_circle', emoji: '🟢' };
    if (st.includes('รับผล')) return { id: "3", label: 'รับผลแล้ว', tag: 'tag-3', icon: 'history', emoji: '📦' };
    return { id: "2", label: 'รอดำเนินการ', tag: 'tag-2', icon: 'pending', emoji: '🟡' };
  }

  function groupByStation(items) {
    return items.reduce((acc, item) => {
      const stKey = item.s || "ไม่ระบุ สภ.";
      if (!acc[stKey]) acc[stKey] = [];
      acc[stKey].push(item);
      return acc;
    }, {});
  }

  function quickFilter(statusId, el) {
    document.querySelectorAll('.filter-pill').forEach(p => p.classList.remove('active'));
    el.classList.add('active');
    document.getElementById('statusFilter').value = statusId;
    runSearch();
  }

  function runSearch() {
    const key = document.getElementById('keyword').value.trim().toLowerCase();
    const station = document.getElementById('stationSelect').value;
    const statFilt = document.getElementById('statusFilter').value;
    
    filteredData = rawData.filter(item => {
      const info = getStatusInfo(item.st);
      const mKey = !key || (item.n.toLowerCase().includes(key) || item.id.toLowerCase().includes(key) || item.s.toLowerCase().includes(key));
      const mStation = !station || item.s === station;
      const mStatus = statFilt === 'all' || info.id === statFilt;
      return mKey && mStation && mStatus;
    });

    if (filteredData.length > 0) {
      generateShareText(key, station);
      renderCards(filteredData);
      document.getElementById('summaryShareSection').classList.remove('d-none');
    } else {
      document.getElementById('summaryShareSection').classList.add('d-none');
      document.getElementById('displayArea').innerHTML = `
        <div class="col-12 text-center py-5">
          <div class="bg-white p-5 rounded-4 shadow-sm d-inline-block border">
            <i class="material-icons-round text-warning" style="font-size: 60px;">sentiment_dissatisfied</i>
            <h5 class="fw-bold mt-3 text-dark">ไม่พบข้อมูลตามเงื่อนไขที่ค้นหา</h5>
            <p class="text-muted small mb-0">ลองเปลี่ยนคำค้นหา หรือเลือกตัวกรองสถานะใหม่อีกครั้งครับ</p>
          </div>
        </div>`;
    }
  }

  function generateShareText(keyword, stationName) {
    const stationHeader = stationName ? `สภ.${stationName}` : "ภาพรวมทุก สภ.";
    const keyHeader = keyword ? ` [คำค้น: "${keyword}"]` : "";
    
    let msg = `🚨 [สรุปสถานะงานตรวจพิสูจน์ พฐ.จว.ตรัง] 🚨\n`;
    msg += `🏛️ หน่วยงาน: ${stationHeader}${keyHeader}\n`;
    msg += `📊 รายการค้นหาพบรวม: ${filteredData.length} รายการ\n`;
    msg += `━━━━━━━━━━━━━━━━━━\n\n`;
    
    const drug = filteredData.filter(d => d.t === "งานยาเสพติด");
    const scene = filteredData.filter(d => d.t === "งานที่เกิดเหตุ");
    const gun = filteredData.filter(d => d.t === "งานทะเบียนปืน");

    if (drug.length > 0) {
      msg += `💊 【 งานตรวจยาเสพติด 】 (${drug.length} รายการ)\n`;
      msg += `──────────────────\n`;
      const groupedDrug = groupByStation(drug);
      Object.keys(groupedDrug).sort().forEach(st => {
        msg += `📍 สภ.${st}\n`;
        groupedDrug[st].forEach((d, idx, arr) => {
          const prefix = idx === arr.length - 1 ? '   └ ' : '   ├ ';
          const info = getStatusInfo(d.st);
          msg += `${prefix}${info.emoji} ${d.id} | ${d.n}\n`;
        });
        msg += `\n`;
      });
    }

    if (scene.length > 0) {
      msg += `🏠 【 งานตรวจสถานที่เกิดเหตุ 】 (${scene.length} รายการ)\n`;
      msg += `──────────────────\n`;
      const groupedScene = groupByStation(scene);
      Object.keys(groupedScene).sort().forEach(st => {
        msg += `📍 สภ.${st}\n`;
        groupedScene[st].forEach((d, idx, arr) => {
          const prefix = idx === arr.length - 1 ? '   └ ' : '   ├ ';
          const info = getStatusInfo(d.st);
          const pongsakorn = d.n.includes('พงส.') ? d.n : `พงส.${d.n}`;
          msg += `${prefix}${info.emoji} ${d.id} | ${pongsakorn}\n`;
        });
        msg += `\n`;
      });
    }

    if (gun.length > 0) {
      msg += `🔫 【 งานทะเบียนปืน 】 (${gun.length} รายการ)\n`;
      msg += `──────────────────\n`;
      const groupedGun = groupByStation(gun);
      Object.keys(groupedGun).sort().forEach(st => {
        msg += `📍 สภ.${st}\n`;
        groupedGun[st].forEach((d, idx, arr) => {
          const prefix = idx === arr.length - 1 ? '   └ ' : '   ├ ';
          const info = getStatusInfo(d.st);
          const pongsakorn = d.n.includes('พงส.') ? d.n : `พงส.${d.n}`;
          msg += `${prefix}${info.emoji} ${d.id} | ${pongsakorn}\n`;
        });
        msg += `\n`;
      });
    }
    
    msg += `━━━━━━━━━━━━━━━━━━\n`;
    msg += `✨ หมายเหตุ: 🟢 เสร็จแล้ว | 🟡 รอดำเนินการ | 📦 รับผลแล้ว\n`;
    msg += `📅 อัปเดตข้อมูล ณ วันที่: ${new Date().toLocaleDateString('th-TH', { year: 'numeric', month: 'short', day: 'numeric' })}`;
    
    currentFormattedMessage = msg;
    document.getElementById('shareTextDisplay').innerText = msg;
    document.getElementById('summaryTotalBadge').innerText = `${filteredData.length} รายการ`;
  }

  function renderCards(data) {
    const area = document.getElementById('displayArea');
    area.innerHTML = '';
    const config = [
      { t: "งานยาเสพติด", i: "medication", label: "ชื่อผู้ต้องหา" },
      { t: "งานที่เกิดเหตุ", i: "location_searching", label: "พนักงานสอบสวน" },
      { t: "งานทะเบียนปืน", i: "security", label: "พนักงานสอบสวน" }
    ];

    config.forEach(conf => {
      const typeData = data.filter(d => d.t === conf.t);
      let html = `<div class="col-lg-4 col-12">
        <div class="column-title">
          <span><i class="material-icons-round align-middle me-2">${conf.i}</i>${conf.t}</span>
          <span class="badge bg-white text-dark rounded-pill px-2.5 py-1 fw-bold">${typeData.length}</span>
        </div>`;

      if (typeData.length > 0) {
        const grouped = groupByStation(typeData);
        Object.keys(grouped).sort().forEach(st => {
          html += `<div class="station-sub-header">
            <span><i class="material-icons-round align-middle me-1 text-secondary" style="font-size:18px;">home</i>สภ.${st}</span>
            <span class="badge bg-secondary bg-opacity-15 text-dark rounded-pill">${grouped[st].length} รายการ</span>
          </div>`;
          
          grouped[st].forEach(item => {
            const info = getStatusInfo(item.st);
            
            const shareMsg = `📢 [รายงานสถานะงาน - พฐ.จว.ตรัง]\n` +
              `━━━━━━━━━━━━━━━\n` +
              `📂 หมวดงาน: ${item.t}\n` +
              `🔢 เลขตรวจ: ${item.id}\n` +
              `👤 ${conf.label}: ${item.n}\n` +
              `🏛️ หน่วยงาน: สภ.${item.s}\n` +
              `📅 วันที่รับเรื่อง: ${item.d}\n` +
              `📌 สถานะ: ${info.emoji} ${item.st}\n` +
              `━━━━━━━━━━━━━━━\n` +
              `🔍 ระบบติดตามงาน พฐ.ตรัง`;

            html += `
              <div class="result-card">
                <a href="#" class="btn-line-mini" onclick="window.open('https://line.me/R/msg/text/?'+encodeURIComponent(\`${shareMsg}\`), '_blank')">
                  <i class="material-icons-round" style="font-size:14px;">share</i> แชร์การ์ด
                </a>
                <div class="case-id-header">${item.id}</div>
                <div class="mb-2">
                  <span style="font-size:0.75rem; color:#94a3b8; display:block; font-weight: 600;">${conf.label}</span>
                  <h6 class="fw-bold m-0 text-dark" style="font-size: 1.02rem;">${item.n}</h6>
                </div>
                <div class="info-row"><i class="material-icons-round text-muted" style="font-size:16px;">calendar_today</i>วันที่รับ: ${item.d}</div>
                <div class="status-badge ${info.tag}">
                  <i class="material-icons-round" style="font-size:15px;">${info.icon}</i> ${item.st}
                </div>
              </div>`;
          });
        });
      } else {
        html += `<div class="text-center p-4 bg-white rounded-4 mb-4 border text-muted small shadow-sm">ไม่มีข้อมูลในหมวดหมู่นี้</div>`;
      }
      
      html += `</div>`;
      area.innerHTML += html;
    });
  }

  function shareToLine() {
    if (!currentFormattedMessage) return;
    if (navigator.share) {
      navigator.share({ title: 'สรุปรายงานการตรวจพิสูจน์ พฐ.จว.ตรัง', text: currentFormattedMessage }).catch(() => {});
    } else {
      copyShareText();
    }
  }

  function copyShareText() {
    if (!currentFormattedMessage) return;
    navigator.clipboard.writeText(currentFormattedMessage).then(() => {
      Notify.copySuccess('คัดลอกข้อความสรุปเรียบร้อยแล้ว!');
    });
  }
